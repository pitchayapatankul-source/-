Located at contracts/TableChainBooking.sol

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract TableChainBooking {
    struct Booking {
        string bookingId;
        address customer;
        address payable restaurantWallet;
        uint256 amount;
        uint256 timestamp;
        PaymentStatus status;
        bool exists;
    }

    event BookingPaid(
        string indexed bookingIdIndexed,
        string bookingId,
        address indexed customerWallet,
        address indexed restaurantWallet,
        uint256 amount,
        uint256 timestamp
    );

    function payBookingFee(
        string calldata bookingId,
        address payable restaurantWallet
    ) external payable {
        require(msg.value > 0, "Booking fee must be > 0");
        require(!bookings[bookingId].exists, "Booking already paid");

        bookings[bookingId] = Booking({
            bookingId: bookingId,
            customer: msg.sender,
            restaurantWallet: restaurantWallet,
            amount: msg.value,
            timestamp: block.timestamp,
            status: PaymentStatus.Paid,
            exists: true
        });

        emit BookingPaid(bookingId, bookingId, msg.sender, restaurantWallet, msg.value, block.timestamp);
    }
}
